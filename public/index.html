01. 조회 기본

조회할 컬럼 → 테이블 → 결과 제한 순으로 작성

기본 조회 · 별칭

SELECT
SELECT
  ID,
  Name AS city_name,
  Population population
FROM city;
AS는 컬럼 별칭에서 생략 가능

중복 제거

DISTINCT
SELECT DISTINCT
  CountryCode,
  District
FROM city;
나열한 컬럼의 조합 전체를 기준으로 중복 제거

상위 N개 · 페이징

LIMIT
-- 처음 10개
SELECT * FROM city ORDER BY ID LIMIT 10;

-- 21번째 행부터 10개 (offset, count)
SELECT * FROM city ORDER BY ID LIMIT 20, 10;
-- 같은 뜻: LIMIT 10 OFFSET 20

02. 조건과 정렬

NULL은 = 대신 IS NULL, 문자열 패턴은 LIKE

비교 · 논리 조건

WHERE
SELECT * FROM city
WHERE Population >= 1000000
  AND CountryCode <> 'KOR'
  AND NOT (District = 'Seoul');
우선순위가 헷갈리면 조건을 괄호로 묶기

목록 · 범위

IN · BETWEEN
WHERE CountryCode IN ('KOR', 'JPN', 'CHN')
  AND Population BETWEEN 100000 AND 500000;
BETWEEN은 양쪽 경계값을 포함

문자열 패턴

LIKE
WHERE Name LIKE 'Kim%'   -- Kim으로 시작
WHERE Name LIKE '%san%' -- san 포함
WHERE Name LIKE '_eoul' -- 임의 1글자
WHERE Name REGEXP '^(New|San)';

NULL 검사

IS NULL
WHERE column_name IS NULL;
WHERE column_name IS NOT NULL;

-- NULL을 대체해 비교
WHERE COALESCE(column_name, '') = '';
= NULL은 사용하지 않음

정렬

ORDER BY
SELECT Name, CountryCode, Population
FROM city
ORDER BY CountryCode ASC, Population DESC, Name ASC;
ASC 오름차순(기본값) · DESC 내림차순 · 결과 순서가 필요하면 반드시 명시

03. 함수와 CASE

자주 쓰는 가공 문법을 한 번에

문자열

TEXT
CONCAT(first_name, ' ', last_name)
LENGTH(Name)          -- 바이트 길이
CHAR_LENGTH(Name)     -- 글자 수
UPPER(Name), LOWER(Name)
TRIM(Name)
SUBSTRING(Name, 1, 3)
REPLACE(Name, 'old', 'new')

숫자 · NULL · 형변환

VALUE
ROUND(price, 2)
CEIL(price), FLOOR(price)
ABS(amount)
COALESCE(nickname, Name, '없음')
NULLIF(value, 0)
CAST(value AS DECIMAL(10,2))
CAST(created_at AS DATE)

날짜

DATE
NOW()                    -- 현재 일시
CURDATE()                -- 현재 날짜
DATE_FORMAT(created_at, '%Y-%m-%d')
DATE_ADD(created_at, INTERVAL 7 DAY)
DATE_SUB(created_at, INTERVAL 1 MONTH)
DATEDIFF(end_date, start_date)
YEAR(created_at), MONTH(created_at)

조건 분기

CASE
CASE
  WHEN Population >= 1000000 THEN '대도시'
  WHEN Population >= 100000  THEN '중도시'
  ELSE '소도시'
END AS city_size

04. 집계와 GROUP BY

그룹 조건은 HAVING, 행 조건은 WHERE

그룹 집계

GROUP BY
SELECT
  CountryCode,
  COUNT(*) AS city_count,
  SUM(Population) AS total_pop,
  AVG(Population) AS avg_pop,
  MAX(Population) AS max_pop
FROM city
WHERE Population > 0
GROUP BY CountryCode
HAVING COUNT(*) >= 10;

조건부 집계

CASE + SUM
SELECT
  CountryCode,
  COUNT(*) AS total_count,
  SUM(CASE WHEN Population >= 1000000
           THEN 1 ELSE 0 END) AS big_count
FROM city
GROUP BY CountryCode;
COUNT(column)은 NULL 제외 · COUNT(*)은 행 전체

05. JOIN

JOIN 조건은 ON에, 결과 필터는 WHERE에 작성

일치하는 행만

INNER JOIN
SELECT ci.Name, co.Name, ci.Population
FROM city AS ci
INNER JOIN country AS co
  ON co.Code = ci.CountryCode;

왼쪽 행 모두 유지

LEFT JOIN
SELECT co.Name, ci.Name
FROM country AS co
LEFT JOIN city AS ci
  ON ci.CountryCode = co.Code
  AND ci.Population >= 1000000;
오른쪽 테이블 조건을 WHERE에 두면 INNER JOIN처럼 될 수 있음

매칭 없는 행 찾기

ANTI JOIN
SELECT co.*
FROM country AS co
LEFT JOIN city AS ci
  ON ci.CountryCode = co.Code
WHERE ci.ID IS NULL;

같은 테이블끼리

SELF JOIN
SELECT e.name, m.name AS manager
FROM employee e
LEFT JOIN employee m
  ON m.id = e.manager_id;

06. 서브쿼리와 CTE

복잡한 쿼리는 WITH로 이름을 붙여 단계별로 읽기

단일값 서브쿼리

SUBQUERY
SELECT Name, Population
FROM city
WHERE Population > (
  SELECT AVG(Population) FROM city
);

존재 여부

EXISTS
SELECT co.*
FROM country co
WHERE EXISTS (
  SELECT 1 FROM city ci
  WHERE ci.CountryCode = co.Code
);

공통 테이블 식

CTE · WITH
WITH country_stats AS (
  SELECT CountryCode, COUNT(*) city_count, SUM(Population) total_pop
  FROM city
  GROUP BY CountryCode
)
SELECT co.Name, cs.city_count, cs.total_pop
FROM country co
JOIN country_stats cs ON cs.CountryCode = co.Code
ORDER BY cs.total_pop DESC;

07. 집합 연산

양쪽 SELECT의 컬럼 수와 호환 가능한 타입을 맞추기

결과 합치기

UNION
-- 중복 제거
SELECT Name, 'CITY' AS type FROM city
UNION
SELECT Name, 'COUNTRY' AS type FROM country;

-- 중복 유지: 일반적으로 더 빠름
SELECT Name FROM table_a
UNION ALL
SELECT Name FROM table_b;
MySQL 8.0.31+는 INTERSECT, EXCEPT 지원

08. 윈도 함수

행을 합치지 않고 순위·누적값·이전값 계산

그룹별 순위

ROW_NUMBER
SELECT
  Name, CountryCode, Population,
  ROW_NUMBER() OVER (
    PARTITION BY CountryCode
    ORDER BY Population DESC
  ) AS rn
FROM city;

누적합 · 이전값

OVER
SUM(amount) OVER (
  PARTITION BY user_id
  ORDER BY paid_at
) AS running_total,
LAG(amount) OVER (
  PARTITION BY user_id
  ORDER BY paid_at
) AS prev_amount
RANK: 공동 순위 뒤 건너뜀 · DENSE_RANK: 건너뛰지 않음

09. INSERT · UPDATE · DELETE

UPDATE와 DELETE 전에는 같은 WHERE로 SELECT 먼저 확인

추가

INSERT
INSERT INTO city
  (Name, CountryCode, District, Population)
VALUES
  ('Seoul', 'KOR', 'Seoul', 9500000),
  ('Busan', 'KOR', 'Busan', 3300000);

조회 결과 추가

INSERT SELECT
INSERT INTO city_backup
  (ID, Name, CountryCode, Population)
SELECT ID, Name, CountryCode, Population
FROM city
WHERE CountryCode = 'KOR';

수정

UPDATE
UPDATE city
SET Population = Population + 1000,
    District = 'Seoul'
WHERE ID = 2331;
WHERE 생략 시 모든 행 수정

삭제

DELETE
DELETE FROM city
WHERE ID = 9999;

TRUNCATE TABLE city_backup;
TRUNCATE는 테이블의 모든 행을 빠르게 제거

있으면 수정, 없으면 추가

UPSERT · MySQL
INSERT INTO scores (user_id, score)
VALUES (1, 90) AS new
ON DUPLICATE KEY UPDATE
  score = new.score;

10. 테이블 · 인덱스

테이블 구조와 제약조건 정의

테이블 생성

CREATE TABLE
CREATE TABLE orders (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  user_id BIGINT NOT NULL,
  order_no VARCHAR(30) NOT NULL UNIQUE,
  amount DECIMAL(12, 2) NOT NULL DEFAULT 0,
  status VARCHAR(20) NOT NULL DEFAULT 'PENDING',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
    ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_orders_user
    FOREIGN KEY (user_id) REFERENCES users(id)
);

구조 변경

ALTER
ALTER TABLE orders
  ADD COLUMN memo VARCHAR(255) NULL;

ALTER TABLE orders
  MODIFY COLUMN memo TEXT;

ALTER TABLE orders
  DROP COLUMN memo;

인덱스

INDEX
CREATE INDEX idx_orders_user_created
  ON orders (user_id, created_at);

CREATE UNIQUE INDEX ux_orders_no
  ON orders (order_no);

DROP INDEX idx_orders_user_created
  ON orders;
복합 인덱스는 보통 왼쪽 컬럼부터 조건에 사용

11. 트랜잭션

여러 변경을 하나의 작업 단위로 처리

시작 · 확정 · 취소

TRANSACTION
START TRANSACTION;

UPDATE accounts SET balance = balance - 10000 WHERE id = 1;
UPDATE accounts SET balance = balance + 10000 WHERE id = 2;

SAVEPOINT transfer_done;
-- ROLLBACK TO transfer_done;
COMMIT; -- 전체 확정
-- ROLLBACK;  -- 전체 취소
